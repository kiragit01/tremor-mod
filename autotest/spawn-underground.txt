# spawn.undergroundOnly: a natural spawn point only in a cave (under the ground, where no light of the sky reaches);
# spawn.darkOnly: only where it is dark enough for monsters. Open sky: none with undergroundOnly, one without; under a
# narrow roof (an overhang) at night: none with undergroundOnly (the user saw one appear there); open sky at night
# with darkOnly: one, at noon: none; under a wide roof (the sky's light gone inside): one with undergroundOnly, also
# with the floor lit, but none once darkOnly is on too.
# Run: gradlew runClientAutotest -Ptremor_autotest_script=spawn-underground.txt
waitchunks
cmd gamemode spectator
cmd time set noon
cmd tremor despawn all
cmd tremor awaken stop
cmd tp @s 300.5 201.0 -899.5 90 30
wait 40
waitchunks
cmd fill 240 198 -960 299 199 -840 minecraft:stone
cmd fill 300 198 -960 360 199 -840 minecraft:stone
cmd fill 240 200 -960 360 200 -840 minecraft:grass_block
cmd fill 240 201 -960 279 206 -840 minecraft:air
cmd fill 280 201 -960 319 206 -840 minecraft:air
cmd fill 320 201 -960 360 206 -840 minecraft:air
cmd gamemode survival
config spawn.darkOnly false
log --- open sky, undergroundOnly: no spawn point
config spawn.undergroundOnly true
cmd tremor spawn natural
cmd tremor despawn all
log --- open sky, anywhere: a spawn point
config spawn.undergroundOnly false
cmd tremor spawn natural
cmd tremor despawn all
log --- under a narrow roof at night (an overhang), undergroundOnly: no spawn point
cmd fill 294 205 -960 306 205 -840 minecraft:stone
cmd time set midnight
wait 60
config spawn.undergroundOnly true
cmd tremor spawn natural
cmd tremor despawn all
cmd fill 294 205 -960 306 205 -840 minecraft:air
log --- open sky at night, darkOnly: a spawn point
config spawn.undergroundOnly false
config spawn.darkOnly true
wait 40
cmd tremor spawn natural
cmd tremor despawn all
log --- open sky at noon, darkOnly: no spawn point
cmd time set noon
wait 20
cmd tremor spawn natural
cmd tremor despawn all
log --- under a wide roof, undergroundOnly: a spawn point
config spawn.darkOnly false
config spawn.undergroundOnly true
cmd fill 240 205 -960 299 205 -840 minecraft:stone
cmd fill 300 205 -960 360 205 -840 minecraft:stone
wait 80
cmd tremor spawn natural
cmd tremor despawn all
log --- under a wide roof, floor lit, undergroundOnly: a spawn point
cmd fill 240 200 -960 299 200 -840 minecraft:glowstone
cmd fill 300 200 -960 360 200 -840 minecraft:glowstone
wait 60
cmd tremor spawn natural
cmd tremor despawn all
log --- under a wide roof, floor lit, undergroundOnly and darkOnly: no spawn point
config spawn.darkOnly true
cmd tremor spawn natural
cmd tremor despawn all
config spawn.darkOnly false
config spawn.undergroundOnly false
cmd fill 240 200 -960 299 200 -840 minecraft:grass_block
cmd fill 300 200 -960 360 200 -840 minecraft:grass_block
cmd fill 240 205 -960 299 205 -840 minecraft:air
cmd fill 300 205 -960 360 205 -840 minecraft:air
quit
