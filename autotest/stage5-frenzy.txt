# Stage 5: the frenzy over a dropped shard. The player throws a shard on the sky platform: nobody can pick it up, a
# little mound swallows it where it lands (gone for good); an entity rises some 24 blocks away and rushes straight
# after the player (fast, tall), who sprints away, and an Awakening starts the moment it reaches them (a build-up too
# short to run from).
# Run: gradlew runClientAutotest -Ptremor_autotest_script=stage5-frenzy.txt
waitchunks
cmd gamemode spectator
cmd time set noon
cmd tremor despawn
cmd tremor awaken stop
cmd tp @s 300.5 201.0 -899.5 90 30
wait 40
waitchunks
cmd fill 240 198 -960 299 199 -840 minecraft:stone
cmd fill 300 198 -960 360 199 -840 minecraft:stone
cmd fill 240 200 -960 360 200 -840 minecraft:grass_block
cmd fill 240 201 -960 279 206 -840 minecraft:air
cmd fill 280 201 -960 319 206 -840 minecraft:air
cmd fill 320 201 -960 360 206 -840 minecraft:air
cmd kill @e[type=!player]
cmd gamemode survival
cmd effect clear @s
cmd clear @s
cmd item replace entity @s weapon.mainhand with tremor:tremor_shard 1
config awakening.craterRadius 0
config awakening.lethal false
hud on
log --- throw the shard
hold drop 2
wait 4
screenshot 10_landed
log --- step onto it: nobody can pick it up
hold forward 8
cmd clear @s tremor:tremor_shard 0
cmd execute if entity @e[type=item,distance=..8]
hold back 8
screenshot 11_mound
wait 30
log --- swallowed: gone for good
cmd execute if entity @e[type=item,distance=..8]
screenshot 13_taken
cmd tremor info
wait 20
screenshot 14_rushing
cmd tremor info
log --- run
hold back+sprint 60
cmd tremor info
cmd tremor awaken stop
wait 40
cmd data get entity @s Dimension
quit
