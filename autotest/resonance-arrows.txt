# Resonance arrows (1.4.0). A flat field, a tremor 80 blocks off. An arrow dropped onto a listener dissolves it; one
# dropped ahead of a running signal soaks it up (the pulse has a reach of 4 blocks); one far from any small creature
# rings through the ground and the tremor hears it; one shot from a bow rings as it flies. Copies tinted red. The
# server log tells what each did.
# Run: gradlew runClientAutotest -Ptremor_autotest_script=resonance-arrows.txt
waitchunks
cmd gamemode spectator
cmd time set noon
cmd tremor despawn all
cmd tremor listener clear
cmd kill @e[type=!player]
cmd gamerule commandModificationBlockLimit 2147483647
cmd tp @s 860.5 215.0 300.5
wait 40
waitchunks
cmd fill 800 188 280 920 199 320 minecraft:dirt
cmd fill 800 200 280 920 200 320 minecraft:grass_block
cmd fill 800 201 280 920 240 320 minecraft:air
cmd kill @e[type=!player]
hud off
tint on
config ambush.enabled false
config listeners.signalSpeed 4
cmd tremor spawn 903 200 300
cmd tremor ai off
cmd tp @s 818.5 201.0 303.5 -60 30
wait 20
# --- 1. dropped onto a listener: it dissolves
cmd tremor listener spawn 823 200 300
wait 30
lookat 823.5 200.5 300.5
screenshot 01_listener
cmd summon minecraft:arrow 823.5 206 300.5 {Tags:["tremor_resonance"],Motion:[0.0,-1.5,0.0]}
wait 7
screenshot 02_pulse
wait 4
screenshot 02b_pulse
wait 4
screenshot 02c_pulse
wait 30
screenshot 03_gone
# --- 2. dropped ahead of a running signal: it is soaked up
cmd tremor listener spawn 823 200 300
cmd tremor ai on
wait 30
cmd summon minecraft:item 823.5 204 299.5 {Item:{id:"minecraft:stone",Count:1b},PickupDelay:32767}
wait 55
cmd summon minecraft:arrow 833.5 206 300.5 {Tags:["tremor_resonance"],Motion:[0.0,-1.5,0.0]}
wait 60
cmd kill @e[type=minecraft:item]
cmd tremor listener clear
# --- 3. far from any small one: it rings through the ground, the tremor hears it
cmd summon minecraft:arrow 870.5 206 300.5 {Tags:["tremor_resonance"],Motion:[0.0,-1.5,0.0]}
wait 20
cmd tremor info
# --- 4. shot from a bow: it rings as it flies
cmd gamemode survival
cmd clear @s
cmd give @s minecraft:bow
cmd give @s tremor:resonance_arrow 4
look -90 -10
hold use 25
wait 6
screenshot 04_flight
wait 6
screenshot 05_flight
wait 60
cmd gamemode spectator
cmd tremor despawn all
config listeners.signalSpeed 7
config ambush.enabled true
tint off
quit
