# A player dies with shards (the user: they could be picked up where the player died): the ground takes them under a
# little mound as it does a thrown one, but no frenzy comes (the player is dead already). Immediate respawn; the place
# is kept loaded and the player comes back to look.
# Run: gradlew runClientAutotest -Ptremor_autotest_script=death-shards.txt
waitchunks
cmd gamerule commandModificationBlockLimit 2147483647
cmd gamerule doImmediateRespawn true
cmd gamemode spectator
cmd time set noon
cmd tremor despawn all
cmd tremor awaken stop
cmd kill @e[type=!player]
cmd tp @s 860.5 202.0 300.5
wait 40
waitchunks
cmd fill 840 194 280 880 199 320 minecraft:dirt
cmd fill 840 200 280 880 200 320 minecraft:grass_block
cmd fill 840 201 280 880 230 320 minecraft:air
cmd forceload add 850 290 870 310
cmd kill @e[type=!player]
cmd tp @s 860.5 201.0 300.5 0 60
cmd gamemode survival
cmd give @s tremor:tremor_shard 3
cmd give @s minecraft:stone 5
wait 10
cmd kill @s
wait 2
cmd execute if entity @e[type=item,x=850,y=195,z=290,dx=20,dy=15,dz=20,nbt={Item:{id:"tremor:tremor_shard"}}]
cmd tp @s 862.5 203.0 300.5 90 40
cmd gamemode spectator
view first
wait 8
screenshot 01_mound
wait 40
screenshot 02_gone
cmd execute unless entity @e[type=item,x=850,y=195,z=290,dx=20,dy=15,dz=20,nbt={Item:{id:"tremor:tremor_shard"}}]
cmd execute if entity @e[type=item,x=850,y=195,z=290,dx=20,dy=15,dz=20,nbt={Item:{id:"minecraft:stone"}}]
cmd tremor info
cmd kill @e[type=item]
cmd forceload remove 850 290 870 310
cmd gamerule doImmediateRespawn false
quit
