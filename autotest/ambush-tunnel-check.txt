# The escape tunnel of a hollow an ambush took the player into (the copy made around the surface over the pit): it
# is there, and walking it to the edge gets the player out. Then the same after an ordinary Awakening on open ground.
waitchunks
cmd gamemode spectator
cmd time set noon
cmd tremor despawn all
cmd tremor awaken stop
cmd tp @s 460.5 201.0 300.5 270 10
wait 40
waitchunks
cmd fill 440 194 280 520 200 320 minecraft:stone
cmd fill 440 200 280 520 200 320 minecraft:grass_block
cmd fill 440 201 280 520 206 320 minecraft:air
config awakening.lethal false
config awakening.craterRadius 0
config hollow.level.breathSeconds 120
log --- an ambush takes the player
cmd gamemode creative
cmd tremor spawn 480 201 300
cmd tremor stage hunting
cmd tremor hold
cmd tremor ambush
wait 40
cmd gamemode survival
cmd tp @s 478.5 201.0 300.5 270 25
wait 240
waitchunks
cmd tremor hollow status
cmd tremor hollow totunnel
wait 10
screenshot 01_ambush_tunnel
hold forward 40
screenshot 02_ambush_in_tunnel
cmd tremor hollow totunnel end
hold forward 60
wait 60
cmd data get entity @s Dimension
screenshot 03_ambush_out
cmd tremor awaken stop
wait 60
log --- an ordinary Awakening on open ground
cmd gamemode survival
cmd tp @s 470.5 201.0 290.5 0 10
config awakening.buildupSeconds 3
cmd tremor spawn 476 201 296
cmd tremor awaken
wait 220
waitchunks
cmd tremor hollow status
cmd tremor hollow totunnel
wait 10
screenshot 04_open_tunnel
hold forward 40
screenshot 05_open_in_tunnel
cmd tremor hollow totunnel end
hold forward 60
wait 60
cmd data get entity @s Dimension
screenshot 06_open_out
cmd tremor awaken stop
quit
