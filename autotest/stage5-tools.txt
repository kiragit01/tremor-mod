# Stage 5: the things made from the shard. A geophone lights a lamp with the entity near; the seismograph in hand points
# at it; boots with "muffled steps" make the steps quiet (hearing debug); the recipes exist; a won node gives shards.
# Run: gradlew runClientAutotest -Ptremor_autotest_script=stage5-tools.txt
waitchunks
cmd gamemode spectator
cmd time set noon
cmd tremor despawn
cmd tp @s 300.5 201.0 300.5 270 10
wait 40
waitchunks
cmd fill 270 198 270 330 200 330 minecraft:stone
cmd fill 270 201 270 330 206 330 minecraft:air
cmd kill @e[type=!player]
cmd gamemode survival
cmd effect clear @s
cmd clear @s
log --- recipes
cmd recipe give @s tremor:seismograph
cmd recipe give @s tremor:geophone
cmd recipe give @s tremor:muffled_steps_book
log --- geophone
cmd setblock 300 200 305 tremor:geophone
cmd setblock 301 200 305 minecraft:redstone_lamp
cmd execute if block 301 200 305 minecraft:redstone_lamp[lit=false]
cmd tremor spawn 312 200 305
cmd tremor ai off
wait 30
cmd execute if block 301 200 305 minecraft:redstone_lamp[lit=true]
cmd data get block 300 200 305
log --- seismograph
cmd give @s tremor:seismograph
cmd give @s tremor:tremor_shard 3
cmd tp @s 300.5 201.0 300.5 270 30
hud on
wait 20
screenshot 10_seismograph_east
cmd tp @s 300.5 201.0 300.5 0 30
wait 20
screenshot 11_seismograph_south
cmd tremor stage hunting
wait 20
screenshot 12_seismograph_hunting
cmd tremor despawn
wait 20
screenshot 13_seismograph_idle
cmd execute if block 301 200 305 minecraft:redstone_lamp[lit=false]
log --- muffled steps
cmd tremor spawn 320 200 290
cmd tremor ai off
cmd tremor debug hearing on
cmd item replace entity @s armor.feet with minecraft:iron_boots
hold forward 30
screenshot 20_steps_plain
cmd item replace entity @s armor.feet with minecraft:iron_boots[enchantments={levels:{"tremor:muffled_steps":1}}]
hold back 30
screenshot 21_steps_muffled
cmd tremor debug hearing off
cmd tremor despawn
log --- shards from a won node
cmd clear @s
config awakening.buildupSeconds 3
cmd tremor awaken
wait 220
waitchunks
cmd tremor hollow outcome victory
wait 120
cmd clear @s tremor:tremor_shard 0
quit
