# The hollow's three fixes, swallowed on a flat field in the sky with a pit and a ravine beside the player:
# 1. the clock starts when the client says the hollow is drawn (see the log: "has the hollow drawn on the client",
#    "is in, the clock starts");
# 2. the pit and the ravine get a scree to climb out by (stills 20, 21; "scree of" in the status);
# 3. standing far out in the field while the closing passes, the player keeps a way out to the way (still 30, the
#    status's "the way out kept open", and the walk to the node from there).
# Run: gradlew runClientAutotest -Ptremor_autotest_script=hollow-fixes.txt
cmd gamerule sendCommandFeedback true
waitchunks
cmd gamerule commandModificationBlockLimit 2147483647
cmd gamemode spectator
cmd time set noon
cmd tremor despawn all
cmd tremor awaken stop
cmd tp @s 300.5 201.0 -899.5 0 10
wait 40
waitchunks
cmd fill 266 186 -934 334 199 -866 minecraft:stone
cmd fill 266 200 -934 334 200 -866 minecraft:grass_block
cmd fill 266 201 -934 334 215 -866 minecraft:air
log --- a pit 5 across and 9 deep east of the player, a ravine 2 wide and 8 deep to the north
cmd fill 306 192 -902 310 200 -898 minecraft:air
cmd fill 296 193 -916 297 200 -906 minecraft:air
cmd kill @e[type=!player]
cmd gamemode survival
cmd effect clear @s
cmd effect give @s minecraft:night_vision infinite 0 true
cmd effect give @s minecraft:instant_health 1 4 true
config client:effects.hollowFogDistance 32
config hollow.level.stillSeconds 60
config hollow.level.breathSeconds 300
config hollow.level.graceSeconds 20
config hollow.level.closeSpeed 1.5
config hollow.level.fillsPerTick 600
hud off
log --- swallowed
cmd tremor hollow enter
wait 120
waitchunks
wait 80
cmd tremor hollow status
log --- the scree out of the pit and the ravine
cmd gamemode spectator
cmd tp @s ~4 ~2 ~-0.5 facing ~8 ~-7 ~-0.5
wait 10
screenshot 20_pit_scree
cmd tp @s ~-8 ~1 ~-3.5 facing ~-8 ~-8 ~-13.5
wait 10
screenshot 21_ravine_scree
cmd tp @s ~4 ~-3 ~4
cmd gamemode survival
log --- far out in the field while the closing passes
cmd tp @s ~-14 ~ ~2 270 15
wait 450
hold left 3
wait 450
cmd tremor hollow status
cmd tremor hollow walk
look 270 5
wait 5
screenshot 30_way_out_kept
look 90 5
wait 5
screenshot 31_closed_behind
cmd tremor hollow leave
wait 100
