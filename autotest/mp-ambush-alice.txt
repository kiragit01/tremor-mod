# The ambush on a server (1.1.2), Alice (tools/mp-test.sh mp-ambush-alice.txt mp-ambush-bob.txt): a tremor lies in
# wait; she steps onto it and sinks into the quicksand while Bob watches; Bob steps into the mud after her and is
# taken too; each is pulled into a hollow of their own; she gets out first (the tremor stays with Bob's ambush), then
# Bob (the tremor goes deep, the ground is as it was). Chat marks keep them in step: each side is waiting for a mark
# before the other one says it.
waitchunks
cmd difficulty easy
cmd time set noon
cmd gamerule doDaylightCycle false
cmd gamemode spectator
cmd fill -20 190 -20 19 199 19 minecraft:stone
cmd fill -20 200 -20 19 210 19 minecraft:air
cmd tp @s 0.5 200 -8.5 0 20
wait 40
waitchunks
cmd tremor despawn all
cmd tremor awaken stop
cmd gamemode creative
waitfor 6000 MARK bob-here
log --- a tremor lies in wait
cmd tremor spawn 0 200 0
cmd tremor stage hunting
cmd tremor hold
cmd tremor ambush
wait 40
cmd say MARK lurking
waitfor 2400 MARK bob-watching
log --- she steps onto it while Bob watches
cmd gamemode survival
cmd say MARK stepping
cmd tp @s 0.5 200 1.5 180 20
wait 160
screenshot a01_in_her_hollow
cmd say MARK alice-in
waitfor 2400 MARK bob-in
log --- she gets out first: the tremor stays with Bob's ambush
cmd tremor awaken stop
wait 100
cmd say MARK alice-out
waitfor 2400 MARK bob-out
wait 40
cmd gamemode creative
cmd tp @s 0.5 201 -6.5 0 30
wait 20
screenshot a02_ground_back
quit
